cmake_minimum_required(VERSION 3.22.1)
project(netdiag_core LANGUAGES CXX)

add_library(netdiag_core STATIC src/probe.cpp)
target_include_directories(netdiag_core PUBLIC include)
target_compile_features(netdiag_core PUBLIC cxx_std_17)
target_compile_options(netdiag_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(netdiag_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    target_link_libraries(netdiag_core PUBLIC android)
endif()

option(NETDIAG_BUILD_TESTS "Build socket integration tests" ON)
if(NETDIAG_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(netdiag_tests tests/probe_tests.cpp)
    target_link_libraries(netdiag_tests PRIVATE netdiag_core Threads::Threads)
    target_compile_options(netdiag_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    foreach(test invalid_input tcp_connect tcp_refused udp_echo udp_timeout udp_malformed cancelled_before cancelled_during fd_cleanup ipv6)
        add_test(NAME ${test} COMMAND netdiag_tests ${test})
        set_tests_properties(${test} PROPERTIES TIMEOUT 10)
    endforeach()
    add_executable(netdiag_echo_server tools/echo_server.cpp)
    target_link_libraries(netdiag_echo_server PRIVATE Threads::Threads)
    target_compile_features(netdiag_echo_server PRIVATE cxx_std_17)
    target_compile_options(netdiag_echo_server PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
