cmake_minimum_required(VERSION 3.22.1)
project(netdiag_jni LANGUAGES CXX)
set(NETDIAG_BUILD_TESTS OFF CACHE BOOL "Build host tests" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../native" "${CMAKE_CURRENT_BINARY_DIR}/core")
add_library(netdiag SHARED jni_bridge.cpp)
target_link_libraries(netdiag PRIVATE netdiag_core)
target_compile_options(netdiag PRIVATE -Wall -Wextra -Wpedantic -Werror -fvisibility=hidden)
target_link_options(netdiag PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
set_target_properties(netdiag PROPERTIES LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
